# **Sinov savollari — Texnik support xodimlari uchun 🧪📋**

Bu sahifa **sinov muddatidagi (probation)** texnik qo'llab-quvvatlash xodimlarining Biznex tizimi bo'yicha bilim darajasini tekshirish uchun mo'ljallangan.

> 💡 **Yo'riqnoma:** Har bir savolga javob bergach, yonidagi katakchani belgilab qo'ying. Sahifaning oxirida nechta savolga javob berganingizni ko'rasiz. Belgilashlar brauzerda saqlanadi (localStorage).

<script is:inline>
(function () {
  const STORAGE_KEY = 'biznex-quiz-test-savollar-v1';

  function init() {
    const article = document.querySelector('main') || document.body;
    const checkboxes = Array.from(article.querySelectorAll('li.task-list-item > input[type="checkbox"]'));
    if (!checkboxes.length) return;

    const progressEl = document.getElementById('biznex-quiz-progress');
    const resetBtn = document.getElementById('biznex-quiz-reset');
    if (!progressEl) return;

    let saved = {};
    try {
      saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
    } catch (e) {
      saved = {};
    }

    checkboxes.forEach((cb, i) => {
      cb.disabled = false;
      cb.style.cursor = 'pointer';
      cb.style.transform = 'scale(1.2)';
      cb.style.marginRight = '0.4rem';
      if (saved[i]) cb.checked = true;
      cb.addEventListener('change', () => {
        saved[String(i)] = cb.checked;
        localStorage.setItem(STORAGE_KEY, JSON.stringify(saved));
        update();
      });
    });

    function update() {
      const total = checkboxes.length;
      const done = checkboxes.filter(c => c.checked).length;
      const percent = total ? Math.round((done / total) * 100) : 0;
      let emoji = '⏳';
      if (percent === 100) emoji = '🎉';
      else if (percent >= 75) emoji = '🚀';
      else if (percent >= 50) emoji = '💪';
      else if (percent >= 25) emoji = '🔥';
      progressEl.innerHTML = emoji + ' Bajarildi: <span style="color: var(--sl-color-accent-high, #818cf8); font-size: 1.25rem;">' + done + '</span> / ' + total + ' (' + percent + '%)';
    }

    if (resetBtn) {
      resetBtn.addEventListener('click', () => {
        if (!confirm('Barcha belgilashlarni tozalashni xohlaysizmi?')) return;
        checkboxes.forEach(cb => { cb.checked = false; });
        localStorage.removeItem(STORAGE_KEY);
        update();
      });
    }

    update();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
</script>
